📁 Google Drive Integration

Set up once. Responses post automatically to a Google Sheet; members can upload files via a shared Drive folder linked to a QR code in the bulletin.

Step 1
Create the Apps Script Webhook
Go to script.google.com → New Project. Paste the code below. Click Deploy → New Deployment → Web App. Set "Who has access" to Anyone. Copy the deployment URL.
// Hold the Line — Worship Response Webhook const SHEET_ID = 'YOUR_GOOGLE_SHEET_ID_HERE'; const SHEET_NAME = 'Responses'; function doPost(e) { try { const data = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.openById(SHEET_ID); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(['Timestamp','Service','Type','Name', 'Contact','Mode','Categories','Notes', 'Crisis','Baptism Public','Raw JSON']); } sheet.appendRow([ new Date().toISOString(), data.service || '', data.type || '', data.name || 'Anonymous', data.contact || '', data.mode || '', (data.categories || []).join('; '), data.notes || '', data.crisis ? 'YES' : '', data.baptismPublic ? 'YES' : '', JSON.stringify(data) ]); return ContentService .createTextOutput(JSON.stringify({success:true})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({success:false,error:err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { return ContentService .createTextOutput('Hold the Line webhook active.') .setMimeType(ContentService.MimeType.TEXT); }
