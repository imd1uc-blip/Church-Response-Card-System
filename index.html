📁 Google Drive Integration

Set up once. Responses post automatically to a Google Sheet; members can upload files via a shared Drive folder linked to a QR code in the bulletin.

Step 1
Create the Apps Script Webhook
Go to script.google.com → New Project. Paste the code below. Click Deploy → New Deployment → Web App. Set "Who has access" to Anyone. Copy the deployment URL.
// Worship Response Webhook — paste into Google Apps Script // Replace the value below with your actual Google Sheet ID var SHEET_ID = 'YOUR_GOOGLE_SHEET_ID_HERE'; var SHEET_NAME = 'Responses'; // doGet handles all incoming requests (avoids all CORS issues) function doGet(e) { try { var p = e.parameter; // If no 'type' param, this is just a connectivity check if (!p || !p.type) { return ContentService .createTextOutput('OK') .setMimeType(ContentService.MimeType.TEXT); } var ss = SpreadsheetApp.openById(SHEET_ID); var sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow([ 'Timestamp','Type','Name','Contact', 'Mode','Categories','Notes','Crisis','Baptism Public' ]); } var categories = p.categories ? p.categories.split('|') : []; sheet.appendRow([ new Date().toISOString(), p.type || '', p.name || 'Anonymous', p.contact || '', p.mode || '', categories.join('; '), p.notes || '', p.crisis === 'true' ? 'YES' : '', p.baptismPublic === 'true' ? 'YES' : '' ]); return ContentService .createTextOutput('OK') .setMimeType(ContentService.MimeType.TEXT); } catch(err) { return ContentService .createTextOutput('ERROR: ' + err.toString()) .setMimeType(ContentService.MimeType.TEXT); } }
