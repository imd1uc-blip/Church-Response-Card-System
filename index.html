📁 Google Drive Integration

Set up once. Members upload directly to Drive; elders access the shared folder. QR in the bulletin sends them straight there.

Step 1
Create the Apps Script Webhook
Copy this code into a new Google Apps Script project (script.google.com → New Project). Deploy it as a Web App set to Anyone access. Copy the deployment URL.
// Hold the Line — Worship Response Webhook // Paste this into Google Apps Script → Deploy as Web App const SHEET_ID = 'YOUR_GOOGLE_SHEET_ID_HERE'; const SHEET_NAME = 'Responses'; function doPost(e) { try { const data = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.openById(SHEET_ID); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(['Timestamp','Service','Type','Name','Contact','Mode', 'Categories','Notes','Crisis','Baptism Public','Raw JSON']); } sheet.appendRow([ new Date().toISOString(), data.service || '', data.type || '', data.name || 'Anonymous', data.contact || '', data.mode || '', (data.categories || []).join('; '), data.notes || '', data.crisis ? 'YES' : '', data.baptismPublic ? 'YES' : '', JSON.stringify(data) ]); return ContentService.createTextOutput( JSON.stringify({success:true}) ).setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput( JSON.stringify({success:false,error:err.toString()}) ).setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { return ContentService.createTextOutput('Hold the Line webhook active.') .setMimeType(ContentService.MimeType.TEXT); }
